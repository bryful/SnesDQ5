cmake_minimum_required(VERSION 3.22)
project(SnesDQ5 VERSION 0.1.18 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Keep local account/workspace paths out of redistributed binaries.
if(MSVC)
  file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" SOURCE_NATIVE_PATH)
  add_compile_options(/experimental:deterministic "/pathmap:${SOURCE_NATIVE_PATH}=SnesDQ5")
  add_link_options("/PDBALTPATH:SnesDQ5.pdb")
else()
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=SnesDQ5")
endif()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)

# Match upstream libretro/Makefile.common instead of maintaining an emulator fork.
set(S9X ${CMAKE_CURRENT_SOURCE_DIR}/third_party/snes9x)
file(READ "${S9X}/libretro/Makefile.common" CORE_MAKEFILE)
string(REGEX MATCHALL "\\$\\(CORE_DIR\\)/[a-zA-Z0-9_./-]+\\.(cpp|c)" CORE_FILES "${CORE_MAKEFILE}")
list(TRANSFORM CORE_FILES REPLACE "\\$\\(CORE_DIR\\)" "${S9X}")
add_library(snes9x STATIC ${CORE_FILES})
target_include_directories(snes9x PUBLIC ${S9X}/libretro PRIVATE ${S9X} ${S9X}/apu ${S9X}/apu/bapu ${S9X}/libretro/libretro-common/include)
target_compile_definitions(snes9x PRIVATE __LIBRETRO__ RIGHTSHIFT_IS_SAR HAVE_STDINT_H ALLOW_CPU_OVERCLOCK)
if(MSVC)
  target_compile_definitions(snes9x PRIVATE __WIN32__ _CRT_SECURE_NO_WARNINGS strcasecmp=_stricmp strncasecmp=_strnicmp)
  target_compile_options(snes9x PRIVATE /wd4244 /wd4267)
else()
  target_compile_definitions(snes9x PRIVATE HAVE_STRINGS_H)
endif()
set(IMGUI ${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui)
add_library(imgui STATIC ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_widgets.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/backends/imgui_impl_sdl2.cpp ${IMGUI}/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(imgui PUBLIC ${IMGUI} ${IMGUI}/backends)
target_link_libraries(imgui PUBLIC SDL2::SDL2-static)
set(APP_SOURCES src/main.cpp src/core.cpp src/storage.cpp)
if(ANDROID)
  add_library(main SHARED ${APP_SOURCES})
  target_link_options(main PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
  add_executable(main ${APP_SOURCES})
  enable_language(RC)
  configure_file(src/windows.rc.in generated/windows.rc @ONLY)
  target_sources(main PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/windows.rc")
  set_target_properties(main PROPERTIES OUTPUT_NAME SnesDQ5)
  target_link_libraries(main PRIVATE SDL2::SDL2main)
endif()
target_link_libraries(main PRIVATE snes9x imgui SDL2::SDL2-static)
target_include_directories(main PRIVATE ${S9X})
if(MSVC)
  target_compile_options(main PRIVATE /utf-8 /W4)
endif()

# Embed the complete notices so they remain readable in Android builds too.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/Snes9x.txt" NOTICE_SNES9X)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/SDL2.txt" NOTICE_SDL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/DearImGui.txt" NOTICE_IMGUI)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/snes_ntsc.txt" NOTICE_NTSC)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/licenses/WiizV.txt" NOTICE_WIIZV)
configure_file(src/licenses.h.in generated/licenses.h @ONLY)
target_include_directories(main PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT ANDROID)
  add_custom_command(TARGET main POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:main>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "$<TARGET_FILE_DIR:main>/README.md")
endif()

option(SNESDQ5_BUILD_REWARD_CHECKS "Build targeted DQ5 reward checks (requires a user-supplied ROM)" OFF)
if(SNESDQ5_BUILD_REWARD_CHECKS AND NOT ANDROID)
  add_executable(input-check tests/input.cpp src/core.cpp src/storage.cpp)
  target_include_directories(input-check PRIVATE ${S9X})
  target_link_libraries(input-check PRIVATE snes9x SDL2::SDL2-static)
  add_executable(reward-check tests/rewards.cpp src/core.cpp src/storage.cpp)
  target_include_directories(reward-check PRIVATE ${S9X})
  target_link_libraries(reward-check PRIVATE snes9x SDL2::SDL2-static)
  if(MSVC)
    target_compile_options(reward-check PRIVATE /utf-8)
  endif()
endif()
